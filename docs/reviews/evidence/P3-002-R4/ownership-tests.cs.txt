using DesktopPet.App;
using DesktopPet.App.Platform;
using DesktopPet.Core.Persistence;
using DesktopPet.Core.Social;
using Microsoft.VisualStudio.TestTools.UnitTesting;

namespace DesktopPet.App.Tests;

/// <summary>
/// P3-002: App-level social integration tests.
///
/// Verifies that the production CreatePetWindow + OnUpdateTick pipeline
/// correctly drives the SocialCoordinator for mutual-notice sessions
/// between two pets. Tests cover:
///   - Normal notice → respond flow
///   - Reject / timeout cancellation
///   - Drag interrupts social session (user input priority)
///   - Pause interrupts social session
///   - Pet removal cancels social session
///   - Social does not delay user grab (head/body press works during social)
///   - Only one session at a time
///   - Social blocks rare-event Evaluate (isSocialActive propagation)
///
/// These tests use the production factory (CreatePetWindow) and the
/// test-only OnUpdateTickTest entry point with injected clocks, following
/// the MultiPetIsolationTests pattern.
/// </summary>
[TestClass]
public class SocialIntegrationTests
{
    /// <summary>
    /// Helper: set up two pets via the production factory with close
    /// positions and controlled clocks. Returns (app, wA, wB).
    /// Both pets are Proactive for predictable social initiation.
    /// </summary>
    private static (App app, MainWindow wA, MainWindow wB) SetupTwoPets(
        double leftA = 100, double topA = 100,
        double leftB = 200, double topB = 100)
    {
        var app = (App?)System.Windows.Application.Current ?? new App();
        var dir = System.IO.Path.Combine(
            System.IO.Path.GetTempPath(), $"Social_{Guid.NewGuid():N}");
        app.TestSetupIsolatedSave(dir);
        app.TestState!.Pets = new List<PetInstanceArchive>
        {
            new() { PetId = "bear-1", Personality = Personality.Proactive, Grievance = 50, WindowLeft = leftA, WindowTop = topA },
            new() { PetId = "bear-2", Personality = Personality.Proactive, Grievance = 50, WindowLeft = leftB, WindowTop = topB },
        };

        app.CreatePetWindow(app.TestState!.Pets[0]);
        app.CreatePetWindow(app.TestState.Pets[1]);
        var wA = app.FindWindowByPetId("bear-1")!;
        var wB = app.FindWindowByPetId("bear-2")!;

        // Set controlled positions (CreatePetWindow may use defaults).
        wA.Left = leftA;
        wA.Top = topA;
        wB.Left = leftB;
        wB.Top = topB;

        return (app, wA, wB);
    }

    /// <summary>
    /// Helper: inject a social coordinator with a high-notice-probability
    /// config and a seeded random so that notice always triggers.
    /// </summary>
    private static SocialCoordinator InjectDeterministicSocial(
        MainWindow wA, MainWindow wB, App app,
        double noticeProbability = 1.0,
        double respondProbability = 1.0,
        double respondDelayMs = 100,
        double noticeDurationMs = 5000,
        double respondDurationMs = 500,
        double minIdleMs = 0,
        double noticeIntervalMs = 0,
        double cooldownMs = 100000,
        double sessionTimeoutMs = 10000)
    {
        var config = new SocialConfig
        {
            NoticeProbability = noticeProbability,
            SlowPacedNoticeProbability = noticeProbability,
            RespondProbability = respondProbability,
            RespondDelayMs = respondDelayMs,
            SlowPacedRespondDelayMs = respondDelayMs,
            NoticeDurationMs = noticeDurationMs,
            RespondDurationMs = respondDurationMs,
            MinIdleMs = minIdleMs,
            NoticeIntervalMs = noticeIntervalMs,
            CooldownMs = cooldownMs,
            SessionTimeoutMs = sessionTimeoutMs,
            NoticeDistanceMaxDip = 1000,
        };
        var coordinator = new SocialCoordinator(config, new Random(42));
        app.SocialCoordinator = coordinator;

        // Re-inject into both windows' StateContext.
        wA.StateContext = new PetStateContext
        {
            PetId = wA.StateContext!.PetId,
            EmotionAccumulator = wA.StateContext.EmotionAccumulator,
            CornerTracking = wA.StateContext.CornerTracking,
            GrievanceExpression = wA.StateContext.GrievanceExpression,
            RuntimeTracker = wA.StateContext.RuntimeTracker,
            RareEventScheduler = wA.StateContext.RareEventScheduler,
            SocialCoordinator = coordinator,
            MarkStateDirty = wA.StateContext.MarkStateDirty,
        };
        wB.StateContext = new PetStateContext
        {
            PetId = wB.StateContext!.PetId,
            EmotionAccumulator = wB.StateContext.EmotionAccumulator,
            CornerTracking = wB.StateContext.CornerTracking,
            GrievanceExpression = wB.StateContext.GrievanceExpression,
            RuntimeTracker = wB.StateContext.RuntimeTracker,
            RareEventScheduler = wB.StateContext.RareEventScheduler,
            SocialCoordinator = coordinator,
            MarkStateDirty = wB.StateContext.MarkStateDirty,
        };

        return coordinator;
    }

    // ════════════════════════════════════════════════════════════════
    // Normal notice → respond flow
    // ════════════════════════════════════════════════════════════════

    /// <summary>
    /// Two idle pets within notice distance: OnUpdateTick should start
    /// a social session (IsSocialActive becomes true).
    /// </summary>
    [STATestMethod]
    public void Social_NormalNotice_SessionStarts()
    {
        var (app, wA, wB) = SetupTwoPets();
        try
        {
            var coordinator = InjectDeterministicSocial(wA, wB, app);

            double now = 1000;
            wA.TestGetElapsedMs = () => now;
            wB.TestGetElapsedMs = () => now;

            // First tick: should start a noticing session.
            wA.OnUpdateTickTest();

            Assert.IsTrue(coordinator.IsSocialActive,
                "Social session should be active after first tick.");
            Assert.IsTrue(wA.IsSocialActive || wB.IsSocialActive,
                "At least one pet should report IsSocialActive.");
        }
        finally
        {
            app.TestCleanup();
        }
    }

    /// <summary>
    /// After notice starts and the respond delay elapses, the session
    /// should transition to Responding phase, then finish and enter
    /// cooldown (IsSocialActive becomes false).
    /// </summary>
    [STATestMethod]
    public void Social_NormalFlow_Notice_Respond_Finish_Cooldown()
    {
        var (app, wA, wB) = SetupTwoPets();
        try
        {
            var coordinator = InjectDeterministicSocial(wA, wB, app,
                respondDelayMs: 100,
                respondDurationMs: 500,
                noticeDurationMs: 5000,
                cooldownMs: 100000);

            double now = 1000;
            wA.TestGetElapsedMs = () => now;
            wB.TestGetElapsedMs = () => now;

            // Start noticing.
            wA.OnUpdateTickTest();
            Assert.IsTrue(coordinator.IsSocialActive,
                "Session should be active (Noticing).");
            Assert.AreEqual(SocialPhase.Noticing, coordinator.CurrentSession.Phase,
                "Should be in Noticing phase.");

            // Advance past respond delay → should transition to Responding.
            now = 1200;
            wA.OnUpdateTickTest();
            Assert.IsTrue(coordinator.IsSocialActive,
                "Session should be active (Responding).");
            Assert.AreEqual(SocialPhase.Responding, coordinator.CurrentSession.Phase,
                "Should be in Responding phase after respond delay.");

            // Advance past respond duration → session should finish.
            now = 1800;
            wA.OnUpdateTickTest();
            Assert.IsFalse(coordinator.IsSocialActive,
                "Session should be finished (not active).");

            // Cooldown should prevent a new session.
            now = 2000;
            wA.OnUpdateTickTest();
            Assert.IsFalse(coordinator.IsSocialActive,
                "Cooldown should prevent a new session.");
        }
        finally
        {
            app.TestCleanup();
        }
    }

    /// <summary>
    /// When RespondProbability is 0, the target rejects the notice.
    /// The session goes through Rejected → Finished and both pets
    /// enter cooldown.
    /// </summary>
    [STATestMethod]
    public void Social_Reject_TargetRejects_SessionEnds()
    {
        var (app, wA, wB) = SetupTwoPets();
        try
        {
            var coordinator = InjectDeterministicSocial(wA, wB, app,
                respondProbability: 0.0,
                respondDelayMs: 100,
                cooldownMs: 100000);

            double now = 1000;
            wA.TestGetElapsedMs = () => now;
            wB.TestGetElapsedMs = () => now;

            // Start noticing.
            wA.OnUpdateTickTest();
            Assert.AreEqual(SocialPhase.Noticing, coordinator.CurrentSession.Phase,
                "Should be in Noticing phase.");

            // Advance past respond delay → should transition to Rejected.
            now = 1200;
            wA.OnUpdateTickTest();
            Assert.AreEqual(SocialPhase.Rejected, coordinator.CurrentSession.Phase,
                "Should be in Rejected phase after respond delay with prob=0.");

            // Next tick: Rejected phase ends immediately → Finished.
            now = 1250;
            wA.OnUpdateTickTest();
            Assert.IsFalse(coordinator.IsSocialActive,
                "Session should end after reject (not active).");

            // Cooldown should prevent immediate restart.
            now = 1300;
            wA.OnUpdateTickTest();
            Assert.IsFalse(coordinator.IsSocialActive,
                "Cooldown should prevent restart after reject.");
        }
        finally
        {
            app.TestCleanup();
        }
    }

    /// <summary>
    /// When a drag starts on either pet during an active social session,
    /// the session is cancelled immediately. The next tick should not
    /// show social active.
    /// </summary>
    [STATestMethod]
    public void Social_DragInterrupts_SessionCancelled()
    {
        var (app, wA, wB) = SetupTwoPets();
        try
        {
            var coordinator = InjectDeterministicSocial(wA, wB, app,
                respondDelayMs: 1000,
                cooldownMs: 100000);

            wB.TestGetElapsedMs = () => now;

            // Start a social session.
            wA.OnUpdateTickTest();
            Assert.IsTrue(coordinator.IsSocialActive, "Session should start.");

            // Start dragging pet A — this should cancel the social session.
            Assert.IsTrue(wA.BeginCapturedPointerInteraction(
                new System.Windows.Point(60, 65),
                new System.Windows.Point(wA.Left + 60, wA.Top + 65)),
                "Body press on pet A should start a drag.");
            Assert.IsTrue(wA.IsDragging, "Pet A should be dragging.");

            // Next tick: social should be cancelled because drag is active.

            wA.OnUpdateTickTest();

                "Social session must be cancelled when drag starts.");

            // Release the drag.
            now = 1060;
            wA.EndCapturedPointerInteraction();
            Assert.IsFalse(wA.IsDragging, "Pet A should not be dragging after release.");
        }
        finally
        {
            app.TestCleanup();
        }
    }

    // ════════════════════════════════════════════════════════════════
    // Personality difference: SlowPaced has lower notice probability
    // ════════════════════════════════════════════════════════════════

    /// <summary>
    /// A SlowPaced pet with SlowPacedNoticeProbability=0 should never
    /// initiate a notice. When paired with a Proactive pet that has
    /// NoticeProbability=1.0, the Proactive pet is always the initiator.
    /// </summary>
    [STATestMethod]
    public void Social_SlowPacedNeverInitiates_ProactiveIsInitiator()
    {
        var app = (App?)System.Windows.Application.Current ?? new App();
        try
        {
            var dir = System.IO.Path.Combine(
                System.IO.Path.GetTempPath(), $"Social_{Guid.NewGuid():N}");
            app.TestSetupIsolatedSave(dir);
            app.TestState!.Pets = new List<PetInstanceArchive>
            {
                new() { PetId = "bear-1", Personality = Personality.Proactive, Grievance = 50, WindowLeft = 100, WindowTop = 100 },
                new() { PetId = "bear-2", Personality = Personality.SlowPaced, Grievance = 50, WindowLeft = 200, WindowTop = 100 },
            };

            app.CreatePetWindow(app.TestState!.Pets[0]);
            app.CreatePetWindow(app.TestState.Pets[1]);
            var wA = app.FindWindowByPetId("bear-1")!;
            var wB = app.FindWindowByPetId("bear-2")!;
            wA.Left = 100; wA.Top = 100;
            wB.Left = 200; wB.Top = 100;

            // Proactive=1.0, SlowPaced=0.0 — only bear-1 can initiate.
            var coordinator = InjectDeterministicSocial(wA, wB, app,
                noticeProbability: 1.0,
                respondDelayMs: 5000,
                cooldownMs: 100000);

            // Override config to set SlowPacedNoticeProbability separately.
            coordinator.Config.SlowPacedNoticeProbability = 0.0;
            coordinator.Config.NoticeProbability = 1.0;

            double now = 1000;
            wA.TestGetElapsedMs = () => now;
            wB.TestGetElapsedMs = () => now;

            // Tick from B's window first — B is SlowPaced, should NOT initiate.
            wB.OnUpdateTickTest();

            // The coordinator deduplicates by nowMs, so the first tick from
            // B already processed. If B could initiate, it would have. But
            // B is SlowPaced with prob=0, so it can't. The session should
            // not start from B.
            // However, since both A and B are tried in TryStartSession,
            // A (Proactive) will be tried first and should initiate.
            // The initiator should be bear-1 (Proactive).
            Assert.IsTrue(coordinator.IsSocialActive,
                "Session should start (Proactive pet initiates).");
            Assert.AreEqual("bear-1", coordinator.CurrentSession.InitiatorId,
                "Initiator must be bear-1 (Proactive), not bear-2 (SlowPaced).");
        }
        finally
        {
            app.TestCleanup();
        }
    }
