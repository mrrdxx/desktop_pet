    // ── Normal notice → respond ──────────────────────────────────────────

    [TestMethod]
    public void Tick_NormalFlow_Notice_Then_Respond()
    {
        var config = new SocialConfig
        {
            NoticeProbability = 1.0,
            RespondProbability = 1.0,
            MinIdleMs = 0,
            NoticeIntervalMs = 0,
            NoticeDistanceMaxDip = 99999,
            NoticeDurationMs = 5000,
            RespondDelayMs = 500,
            RespondDurationMs = 2000,
            CooldownMs = 99999,
            SessionTimeoutMs = 99999
        };
        var coord = new SocialCoordinator(config, new Random(42));
        var petA = IdlePet("bear-1", Personality.Proactive, cx: 100);
        var petB = IdlePet("bear-2", Personality.Proactive, cx: 200);

        // Tick 1: should start Noticing.
        var r1 = coord.Tick(5000, petA, petB);
        Assert.IsTrue(r1.IsSocialActive);
        Assert.IsTrue(r1.SessionStarted);
        Assert.AreEqual(SocialPhase.Noticing, r1.Session.Phase);
        Assert.AreEqual(SocialExpression.Notice, r1.InitiatorExpression);

        // Tick 2: before respond delay — still Noticing.
        var r2 = coord.Tick(5100, petA, petB);
        Assert.IsTrue(r2.IsSocialActive);
        Assert.AreEqual(SocialPhase.Noticing, r2.Session.Phase);

        // Tick 3: after respond delay — should transition to Responding.
        var r3 = coord.Tick(5600, petA, petB);
        Assert.IsTrue(r3.IsSocialActive);
        Assert.AreEqual(SocialPhase.Responding, r3.Session.Phase);
        Assert.AreEqual(SocialExpression.Respond, r3.TargetExpression);

        // Tick 4: before respond duration ends — still Responding.
        var r4 = coord.Tick(7000, petA, petB);
        Assert.IsTrue(r4.IsSocialActive);
        Assert.AreEqual(SocialPhase.Responding, r4.Session.Phase);

        // Tick 5: after respond duration — session finishes.
        var r5 = coord.Tick(8200, petA, petB);
        Assert.IsFalse(r5.IsSocialActive);
        Assert.IsTrue(r5.SessionEnded);
        Assert.AreEqual(SocialPhase.Finished, r5.Session.Phase);
    }

    [TestMethod]
    public void Cooldown_Prevents_ImmediateRestart()
    {
        var config = new SocialConfig
        {
            NoticeProbability = 1.0, RespondProbability = 1.0,
            MinIdleMs = 0, NoticeIntervalMs = 0, NoticeDistanceMaxDip = 99999,
            NoticeDurationMs = 5000, RespondDelayMs = 500,
            RespondDurationMs = 500, CooldownMs = 10000,
            SessionTimeoutMs = 99999
        };
        var coord = new SocialCoordinator(config, new Random(42));
        var petA = IdlePet("bear-1", cx: 100);
        var petB = IdlePet("bear-2", cx: 200);

        // Start and complete a full session.
        coord.Tick(5000, petA, petB);  // Noticing
        coord.Tick(5600, petA, petB);  // Responding
        var end = coord.Tick(6200, petA, petB); // Finished + cooldown starts
        Assert.IsTrue(end.SessionEnded);

        // Immediately after — should be in cooldown, not start new.
        var r = coord.Tick(6300, petA, petB);
        Assert.IsFalse(r.IsSocialActive);
        Assert.IsFalse(r.SessionStarted);

        // After cooldown — should be able to start again.
        var r2 = coord.Tick(16400, petA, petB);
        Assert.IsTrue(r2.IsSocialActive);
        Assert.IsTrue(r2.SessionStarted);
    }

    [TestMethod]
    public void NoticeInterval_Prevents_Spam()
    {
        var config = new SocialConfig
        {
            NoticeProbability = 1.0,
            SlowPacedNoticeProbability = 0.0, // petB (SlowPaced) never initiates
            RespondProbability = 0.0,
            MinIdleMs = 0, NoticeIntervalMs = 5000, NoticeDistanceMaxDip = 99999,
            NoticeDurationMs = 500, RespondDelayMs = 99999,
            CooldownMs = 0, SessionTimeoutMs = 99999
        };
        var coord = new SocialCoordinator(config, new Random(42));
        var petA = IdlePet("bear-1", Personality.Proactive, cx: 100);
        var petB = IdlePet("bear-2", Personality.SlowPaced, cx: 200);

        // First attempt by petA succeeds.
        var r1 = coord.Tick(5000, petA, petB);
        Assert.IsTrue(r1.SessionStarted);

        // Let it expire (noticing phase ends).
        var r2 = coord.Tick(5600, petA, petB);
        Assert.IsTrue(r2.SessionEnded);

        // Immediately try again — petA blocked by NoticeInterval, petB never initiates.
        var r3 = coord.Tick(5700, petA, petB);
        Assert.IsFalse(r3.IsSocialActive);

        // After interval — petA can try again.
        var r4 = coord.Tick(10800, petA, petB);
        Assert.IsTrue(r4.SessionStarted);
    }

    [TestMethod]
    public void SlowPaced_LowerNoticeProbability_RarelyInitiates()
    {
        // With SlowPaced probability 0.0, should never initiate.
        var config = new SocialConfig
        {
            NoticeProbability = 1.0,
            SlowPacedNoticeProbability = 0.0,
            MinIdleMs = 0, NoticeIntervalMs = 0, NoticeDistanceMaxDip = 99999,
            CooldownMs = 0, SessionTimeoutMs = 99999
        };
        var coord = new SocialCoordinator(config, new Random(42));
        var petA = IdlePet("bear-1", Personality.SlowPaced, cx: 100);
        var petB = IdlePet("bear-2", Personality.Proactive, cx: 200);

        // Try 20 ticks — SlowPaced should never initiate, Proactive might.
        bool anyStart = false;
        for (int i = 0; i < 20; i++)
        {
            var r = coord.Tick(5000 + i * 100, petA, petB);
            if (r.SessionStarted)
            {
                anyStart = true;
                // If session started, it should be Proactive (bear-2) as initiator.
                Assert.AreEqual("bear-2", r.Session.InitiatorId);
                break;
            }
        }
        // Proactive should have initiated at some point.
        Assert.IsTrue(anyStart);
    }

    // ── Both can initiate ────────────────────────────────────────────────

    [TestMethod]
    public void Either_Pet_Can_Be_Initiator()
    {
        // petA has a long notice interval, petB has 0 — after petA's first attempt,
        // only petB can initiate the next session.
        var config = new SocialConfig
        {
            NoticeProbability = 1.0, RespondProbability = 0.0,
            MinIdleMs = 0, NoticeIntervalMs = 0, NoticeDistanceMaxDip = 99999,
            NoticeDurationMs = 100, RespondDelayMs = 99999,
            CooldownMs = 0, SessionTimeoutMs = 99999
        };
        var coord = new SocialCoordinator(config, new Random(42));
        var petA = IdlePet("bear-1", cx: 100);
        var petB = IdlePet("bear-2", cx: 200);

        // First tick: petA (first checked) initiates.
        var r1 = coord.Tick(5000, petA, petB);
        Assert.IsTrue(r1.SessionStarted);
        Assert.AreEqual("bear-1", r1.Session.InitiatorId);

        // Let it finish (noticing duration expires).
        var r2 = coord.Tick(5200, petA, petB);
        Assert.IsTrue(r2.SessionEnded);

        // Next tick: petA already has lastAttempt=5000, petB has lastAttempt=0.
        // Both have NoticeIntervalMs=0, so petA (first checked) initiates again.
        var r3 = coord.Tick(5300, petA, petB);
        Assert.IsTrue(r3.SessionStarted);
        Assert.AreEqual("bear-1", r3.Session.InitiatorId);

        // Now swap order: pass petB first. petB should be able to initiate.
        var r4 = coord.Tick(5500, petB, petA);
        // petB is now checked first, and has lastAttempt=0 (never tried as initiator).
        // But session from r3 ended at 5300+100=5400, cooldown=0 so ok.
        if (r4.SessionStarted)
        {
            // petB should be the initiator when passed first.
            Assert.AreEqual("bear-2", r4.Session.InitiatorId);
        }
    }

    // ── P3-003: Seeking state machine ───────────────────────────────────

    /// <summary>
    /// Helper: create an idle pet on the bottom edge of a monitor.
    /// </summary>
    private static PetSocialState SeekablePet(
        string petId, double cx, double bottomEdgeY = 800,
        string? monitorId = "DISPLAY1", double lastActivity = 0,
        bool onBottomEdge = true) =>
        new(petId, Personality.Proactive, cx, bottomEdgeY,
            isDragging: false, isPettingCandidate: false,
            isPhysicsActive: false, isExpressingGrievance: false,
            isProbing: false, isPaused: false, lastActivity,
            monitorId: monitorId, bottomEdgeY: bottomEdgeY,
            isOnBottomEdge: onBottomEdge);

    private static SocialConfig SeekConfig(
        double reactDelay = 100, double stepDip = 4,
        double stopDist = 120, double maxDist = 1200,
        double timeout = 20000, double cooldown = 20000,
        double bottomTol = 8) => new()
        {
            // Disable mutual-notice so seeking is the only path.
            NoticeProbability = 0.0,
            SlowPacedNoticeProbability = 0.0,
            MinIdleMs = 0,
            NoticeIntervalMs = 0,
            NoticeDistanceMaxDip = 99999,
            CooldownMs = 0,
            SessionTimeoutMs = 99999,
            SeekReactDelayMs = reactDelay,
            SeekStepDip = stepDip,
            SeekStopDistanceDip = stopDist,
            SeekMaxDistanceDip = maxDist,
            SeekTimeoutMs = timeout,
            SeekCooldownMs = cooldown,
            SeekBottomToleranceDip = bottomTol,
        };

    [TestMethod]
    public void Seek_BothIdleSameBottom_StartsSeeking()
    {
        var coord = new SocialCoordinator(SeekConfig(), new Random(42));
        // A was recently active (dragged), B is idle. Both on same bottom edge.
        var petA = SeekablePet("bear-1", cx: 500, lastActivity: 5000);
        var petB = SeekablePet("bear-2", cx: 200, lastActivity: 0);

        // P3-003 R1 F-01: Must call NotifyPetReleased to establish a seeking candidate.
        coord.NotifyPetReleased("bear-1", 5000, wasThrown: false);

        // After react delay (5000 + 100 = 5100).

    {
        var coord = new SocialCoordinator(SeekConfig(
            reactDelay: 0, stepDip: 50, stopDist: 100, cooldown: 99999),
            new Random(42));
        // B at 200, A at 500. Distance = 300. B seeks A.
        var petA = SeekablePet("bear-1", cx: 500, lastActivity: 1000);
        var petB = SeekablePet("bear-2", cx: 200, lastActivity: 0);

        // P3-003 R1 F-01: Establish release candidate.
        coord.NotifyPetReleased("bear-1", 1000, wasThrown: false);

        // Start seeking.
        var r1 = coord.Tick(2000, petA, petB);
        Assert.AreEqual(SocialPhase.Seeking, r1.Session.Phase);
        Assert.AreEqual("bear-2", r1.SeekerId);
        Assert.IsTrue(r1.SeekStepDelta!.Value > 0);

        // Simulate B moving: update petB's position.
        // P3-003 R1 F-02: Steps are time-based. Use 33ms intervals.
        var movedB = SeekablePet("bear-2", cx: 250, lastActivity: 0);
        var r2 = coord.Tick(2033, petA, movedB);
        Assert.AreEqual(SocialPhase.Seeking, r2.Session.Phase);

        // Continue stepping until within stop distance.
        // Distance = 500 - 250 = 250, step = min(50, 250-100) = 50.
        // After 4 more steps of 50: 250→300→350→400→450, dist=50 < 100 → Arrived.
        var currentX = 250.0;
        for (int i = 0; i < 10; i++)
        {
            currentX += 50;
            var moved = SeekablePet("bear-2", cx: currentX, lastActivity: 0);
            var r = coord.Tick(2066 + i * 33, petA, moved);
            if (r.Session.Phase == SocialPhase.Arrived)
            {
                // Arrived: step delta should be 0 (stop).
                Assert.AreEqual(0, r.SeekStepDelta);
                return;
            }
        }

        Assert.Fail("Should have arrived within 10 steps.");
    }

        var b = SeekablePet("bear-2", cx: 200, lastActivity: 0);
        var r = c.Tick(5000, a, b);
        Assert.AreEqual(SocialPhase.ChaseInviting, r.Session.Phase);
        var leader = r.Session.ChaseLeaderId == a.PetId ? a : b;
        var follower = leader.PetId == a.PetId ? b : a;
        Assert.IsTrue((leader.CenterX - follower.CenterX) * r.Session.ChaseDirection!.Value > 0,
            $"Leader {leader.PetId} at {leader.CenterX}, follower at {follower.CenterX}, direction {r.Session.ChaseDirection}.");
    }

    [TestMethod]
    public void Reviewer_PendingReleaseGetsItsResponseWindowBeforeChase()
    {
        var config = ChaseConfig();
        config.SeekReactDelayMs = 1500;
        var c = new SocialCoordinator(config, new Random(42));
        var a = SeekablePet("bear-1", cx: 200, lastActivity: 0);
        var b = SeekablePet("bear-2", cx: 400, lastActivity: 0);
        c.NotifyPetReleased(a.PetId, 5000, wasThrown: false);
        Assert.AreEqual(SocialPhase.Idle, c.Tick(5100, a, b).Session.Phase);
        Assert.AreEqual(SocialPhase.Seeking, c.Tick(6500, a, b).Session.Phase);
    }

    [TestMethod]
    public void Reviewer_ChaseCooldownMustNotBlockFreshRelease()
    {
        var config = ChaseConfig();
        config.SeekReactDelayMs = 1500;
        var c = new SocialCoordinator(config, new Random(42));
        var a = SeekablePet("bear-1", cx: 200, lastActivity: 0);
        var b = SeekablePet("bear-2", cx: 400, lastActivity: 0);
        Assert.AreEqual(SocialPhase.ChaseInviting, c.Tick(5000, a, b).Session.Phase);
        c.ForceCancel(CancelReason.DragStarted, 5100);
        c.NotifyPetReleased(a.PetId, 5500, wasThrown: false);
        Assert.AreEqual(SocialPhase.Seeking, c.Tick(7100, a, b).Session.Phase);
    }
